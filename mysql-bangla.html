MySQL কী?

MySQL একটি Relational Database Management System (RDBMS)। এটি ডেটাকে সারি (rows) ও কলাম (columns) আকারে টেবিলে সংরক্ষণ করে। Facebook, Twitter, YouTube সহ বড় বড় কোম্পানি MySQL ব্যবহার করে।

💡 কেন MySQL?
✅ বিনামূল্যে ও ওপেন সোর্স
✅ দ্রুত ও নির্ভরযোগ্য
✅ সব প্ল্যাটফর্মে চলে (Windows, Linux, Mac)
✅ PHP, Python, Java সব ভাষার সাথে কাজ করে

MySQL কোন ধরনের ডেটাবেস সিস্টেম?

🟢 শুরু

ডেটাবেস তৈরি ও পরিচালনা

এই পাঠে আমরা ডেটাবেস তৈরি করা, দেখা, সিলেক্ট করা এবং মুছে ফেলা শিখবো।

ডেটাবেস তৈরি করো

SQL
-- নতুন ডেটাবেস তৈরি করো
CREATE DATABASE school_db;

-- যদি আগে থেকে না থাকে তাহলে তৈরি করো
CREATE DATABASE IF NOT EXISTS school_db;

-- বাংলা সাপোর্টের জন্য charset সেট করো
CREATE DATABASE school_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

ডেটাবেস দেখো ও ব্যবহার করো

SQL
-- সব ডেটাবেস দেখো
SHOW DATABASES;

-- একটি ডেটাবেস ব্যবহার করো
USE school_db;

-- বর্তমান ডেটাবেস জানো
SELECT DATABASE();

-- ডেটাবেস মুছে ফেলো (সাবধান!)
DROP DATABASE school_db;
DROP DATABASE IF EXISTS school_db;
⚠️ সতর্কতা: DROP DATABASE চালালে সব ডেটা চিরতরে মুছে যাবে। সাবধানে ব্যবহার করো!

🟢 শুরু

টেবিল তৈরি করা

টেবিল হলো ডেটাবেসের মূল কাঠামো। এখানে ডেটা সারিবদ্ধভাবে সংরক্ষিত হয়।

টেবিল তৈরি করো

SQL
USE school_db;

CREATE TABLE students (
  id         INT AUTO_INCREMENT PRIMARY KEY,
  name       VARCHAR(100) NOT NULL,
  roll       INT UNIQUE NOT NULL,
  class      VARCHAR(20),
  marks      DECIMAL(5,2) DEFAULT 0.00,
  dob        DATE,
  is_active  BOOLEAN DEFAULT TRUE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

Constraint মানে কী?

Constraintঅর্থ
PRIMARY KEYটেবিলের প্রতিটি সারির অনন্য পরিচয়
NOT NULLএই কলাম খালি রাখা যাবে না
UNIQUEএকই মান দুইবার থাকতে পারবে না
DEFAULTকোনো মান না দিলে এটি ব্যবহার হবে
AUTO_INCREMENTস্বয়ংক্রিয়ভাবে ১ বাড়বে
FOREIGN KEYঅন্য টেবিলের সাথে সংযোগ

টেবিল দেখো ও পরিবর্তন করো

SQL
-- সব টেবিল দেখো
SHOW TABLES;

-- টেবিলের কাঠামো দেখো
DESCRIBE students;

-- নতুন কলাম যোগ করো
ALTER TABLE students ADD email VARCHAR(150);

-- কলামের নাম পরিবর্তন করো
ALTER TABLE students
  RENAME COLUMN marks TO total_marks;

-- কলাম মুছে ফেলো
ALTER TABLE students DROP COLUMN email;

-- টেবিল মুছে ফেলো
DROP TABLE students;

ডেটা ইনসার্ট করা (INSERT)

টেবিলে ডেটা যোগ করতে INSERT INTO ব্যবহার করা হয়।

একটি সারি যোগ করো

SQL
-- সব কলামে ডেটা দাও
INSERT INTO students (name, roll, class, marks, dob)
VALUES ('আরিফ হোসেন', 101, 'দশম', 87.50, '2008-03-15');

-- id এবং created_at স্বয়ংক্রিয়ভাবে যোগ হবে

একসাথে একাধিক সারি যোগ করো

SQL
INSERT INTO students (name, roll, class, marks)
VALUES
  ('সুমাইয়া বেগম', 102, 'দশম', 92.00),
  ('রাকিব হাসান',  103, 'নবম',  78.50),
  ('তানভীর আলম',  104, 'নবম',  85.00),
  ('নিশাত জাহান', 105, 'দশম', 95.50);
💡 টিপস: যে কলামে DEFAULT বা AUTO_INCREMENT আছে সেটি ইনসার্টে বাদ দেওয়া যায়। NOT NULL কলাম অবশ্যই দিতে হবে।

ডেটা দেখো

SQL
-- সব ডেটা দেখো
SELECT * FROM students;

-- শেষ ইনসার্টের ID জানো
SELECT LAST_INSERT_ID();

🧠 কুইজ

একসাথে ৩টি সারি ইনসার্ট করতে কোন পদ্ধতি সবচেয়ে ভালো?

🔵 মধ্যবর্তী

SELECT ও WHERE দিয়ে ডেটা খোঁজো

SELECT হলো MySQL এর সবচেয়ে বেশি ব্যবহৃত কমান্ড। WHERE দিয়ে শর্ত দিয়ে নির্দিষ্ট ডেটা খোঁজা যায়।

SELECT এর মূল ব্যবহার

SQL
-- সব কলাম দেখো
SELECT * FROM students;

-- নির্দিষ্ট কলাম দেখো
SELECT name, roll, marks FROM students;

-- কলামের নাম পরিবর্তন করে দেখাও (Alias)
SELECT name AS 'ছাত্রের নাম', marks AS 'নম্বর'
FROM students;

WHERE দিয়ে শর্ত দাও

SQL
-- নির্দিষ্ট ক্লাসের ছাত্র
SELECT * FROM students
WHERE class = 'দশম';

-- ৮০ এর উপরে নম্বর পেয়েছে
SELECT name, marks FROM students
WHERE marks > 80;

-- AND / OR ব্যবহার
SELECT * FROM students
WHERE class = 'দশম' AND marks > 85;

-- BETWEEN ব্যবহার
SELECT * FROM students
WHERE marks BETWEEN 70 AND 90;

-- LIKE ব্যবহার (আংশিক মিলানো)
SELECT * FROM students
WHERE name LIKE '%হাসান%';

-- IN ব্যবহার
SELECT * FROM students
WHERE roll IN (101, 103, 105);

-- NULL চেক
SELECT * FROM students
WHERE dob IS NULL;

তুলনামূলক অপারেটর

অপারেটরঅর্থউদাহরণ
=সমানmarks = 90
!= বা <>সমান নয়class != 'নবম'
>বড়marks > 80
<ছোটroll < 110
LIKE '%x%'আংশিক মিলানোname LIKE '%আলম%'
BETWEEN a AND bপরিসরের মধ্যেmarks BETWEEN 60 AND 80

🔵 মধ্যবর্তী

UPDATE ও DELETE

ডেটা পরিবর্তন করতে UPDATE এবং মুছতে DELETE ব্যবহার করা হয়।

UPDATE — ডেটা পরিবর্তন করো

SQL
-- একটি নির্দিষ্ট সারি আপডেট করো
UPDATE students
SET marks = 95.00
WHERE roll = 101;

-- একাধিক কলাম আপডেট করো
UPDATE students
SET marks = 88.00, class = 'একাদশ'
WHERE id = 3;

-- সবার নম্বরে ৫ যোগ করো
UPDATE students
SET marks = marks + 5
WHERE class = 'দশম';
⚠️ গুরুত্বপূর্ণ: UPDATE এ সবসময় WHERE ব্যবহার করো! না হলে সব সারির ডেটা পরিবর্তন হয়ে যাবে।

DELETE — ডেটা মুছে ফেলো

SQL
-- একটি নির্দিষ্ট সারি মুছো
DELETE FROM students
WHERE roll = 104;

-- শর্ত অনুযায়ী মুছো
DELETE FROM students
WHERE marks < 50;

-- সব ডেটা মুছো (কাঠামো থাকবে)
TRUNCATE TABLE students;

🔵 মধ্যবর্তী

ORDER BY ও LIMIT

ফলাফল সাজানো এবং নির্দিষ্ট সংখ্যক সারি দেখানোর জন্য ORDER BY ও LIMIT ব্যবহার করা হয়।

ORDER BY — ফলাফল সাজাও

SQL
-- নম্বর অনুযায়ী ছোট থেকে বড় (ASC)
SELECT * FROM students
ORDER BY marks ASC;

-- নম্বর অনুযায়ী বড় থেকে ছোট (DESC)
SELECT * FROM students
ORDER BY marks DESC;

-- নাম অনুযায়ী বর্ণানুক্রমে
SELECT * FROM students
ORDER BY name ASC;

-- একাধিক কলামে সাজাও
SELECT * FROM students
ORDER BY class ASC, marks DESC;

LIMIT — নির্দিষ্ট সংখ্যক সারি দেখাও

SQL
-- প্রথম ৫টি ছাত্র
SELECT * FROM students LIMIT 5;

-- সর্বোচ্চ নম্বর পাওয়া ৩ জন
SELECT name, marks FROM students
ORDER BY marks DESC
LIMIT 3;

-- OFFSET: ৬ নম্বর থেকে ১০ নম্বর সারি (Pagination)
SELECT * FROM students
LIMIT 5 OFFSET 5;
-- অথবা সংক্ষেপে: LIMIT 5, 5

-- DISTINCT: অনন্য মান দেখাও
SELECT DISTINCT class FROM students;
📌 Pagination উদাহরণ:
Page 1: LIMIT 10 OFFSET 0
Page 2: LIMIT 10 OFFSET 10
Page 3: LIMIT 10 OFFSET 20

সর্বনিম্ন নম্বর পাওয়া ১ জন ছাত্র দেখাতে কোনটি সঠিক?

🔵 মধ্যবর্তী

ফাংশন ও Aggregate

MySQL এ বিভিন্ন বিল্ট-ইন ফাংশন আছে যা ডেটা প্রক্রিয়া করতে সাহায্য করে।

Aggregate ফাংশন (গণনা করার জন্য)

SQL
-- মোট ছাত্র সংখ্যা
SELECT COUNT(*) AS মোট_ছাত্র FROM students;

-- সর্বোচ্চ নম্বর
SELECT MAX(marks) AS সর্বোচ্চ FROM students;

-- সর্বনিম্ন নম্বর
SELECT MIN(marks) AS সর্বনিম্ন FROM students;

-- নম্বরের যোগফল
SELECT SUM(marks) AS মোট_নম্বর FROM students;

-- গড় নম্বর
SELECT AVG(marks) AS গড়_নম্বর FROM students;

-- একসাথে সব দেখাও
SELECT
  COUNT(*) AS মোট,
  MAX(marks) AS সর্বোচ্চ,
  MIN(marks) AS সর্বনিম্ন,
  ROUND(AVG(marks), 2) AS গড়
FROM students;

String ফাংশন

SQL
SELECT
  UPPER(name),        -- বড় হাতে
  LOWER(name),        -- ছোট হাতে
  LENGTH(name),       -- অক্ষর সংখ্যা
  CONCAT(name, ' - ', class),  -- যোগ করো
  SUBSTRING(name, 1, 5),       -- অংশ নাও
  TRIM(name)           -- আগে-পরের ফাঁকা সরাও
FROM students;

Date ফাংশন

SQL
SELECT
  NOW(),              -- এখনকার তারিখ ও সময়
  CURDATE(),          -- আজকের তারিখ
  YEAR(dob),          -- জন্মের বছর
  MONTH(dob),         -- জন্মের মাস
  DATEDIFF(CURDATE(), dob) / 365 AS বয়স
FROM students;
🔵 মধ্যবর্তী

GROUP BY ও HAVING

GROUP BY দিয়ে ডেটা দলভিত্তিক বিশ্লেষণ করা যায়। HAVING দিয়ে গ্রুপের উপর শর্ত দেওয়া হয়।

GROUP BY — দলভিত্তিক বিশ্লেষণ

SQL
-- ক্লাসভিত্তিক ছাত্র সংখ্যা
SELECT class, COUNT(*) AS ছাত্র_সংখ্যা
FROM students
GROUP BY class;

-- ক্লাসভিত্তিক গড় নম্বর
SELECT
  class,
  COUNT(*) AS সংখ্যা,
  ROUND(AVG(marks), 2) AS গড়_নম্বর,
  MAX(marks) AS সর্বোচ্চ
FROM students
GROUP BY class
ORDER BY গড়_নম্বর DESC;

HAVING — গ্রুপে শর্ত দাও

SQL
-- যে ক্লাসে ৩ বা বেশি ছাত্র আছে
SELECT class, COUNT(*) AS সংখ্যা
FROM students
GROUP BY class
HAVING COUNT(*) >= 3;

-- যে ক্লাসের গড় নম্বর ৮০ এর উপরে
SELECT class, AVG(marks) AS গড়
FROM students
GROUP BY class
HAVING AVG(marks) > 80;
WHERE বনাম HAVING:
📌 WHERE — সারি ফিল্টার করে (GROUP BY এর আগে)
📌 HAVING — গ্রুপ ফিল্টার করে (GROUP BY এর পরে)
🔴 উন্নত

JOIN — টেবিল সংযুক্তি

JOIN দিয়ে একাধিক টেবিলের ডেটা একসাথে দেখা যায়। এটি MySQL এর সবচেয়ে গুরুত্বপূর্ণ বিষয়গুলোর একটি।

উদাহরণ টেবিল তৈরি করো

SQL
-- শিক্ষক টেবিল
CREATE TABLE teachers (
  id      INT AUTO_INCREMENT PRIMARY KEY,
  name    VARCHAR(100),
  subject VARCHAR(50)
);

-- students টেবিলে teacher_id যোগ করো
ALTER TABLE students
  ADD COLUMN teacher_id INT,
  ADD FOREIGN KEY (teacher_id) REFERENCES teachers(id);

INNER JOIN — উভয় টেবিলে মিলে এমন

SQL
SELECT
  s.name AS ছাত্র,
  s.marks,
  t.name AS শিক্ষক,
  t.subject
FROM students s
INNER JOIN teachers t
  ON s.teacher_id = t.id;

LEFT JOIN ও RIGHT JOIN

SQL
-- LEFT JOIN: সব ছাত্র দেখাও (শিক্ষক না থাকলেও)
SELECT s.name, t.name AS শিক্ষক
FROM students s
LEFT JOIN teachers t ON s.teacher_id = t.id;

-- RIGHT JOIN: সব শিক্ষক দেখাও (ছাত্র না থাকলেও)
SELECT s.name, t.name AS শিক্ষক
FROM students s
RIGHT JOIN teachers t ON s.teacher_id = t.id;

JOIN এর ধরন

JOINফলাফল
INNER JOINউভয় টেবিলে মিলে এমন সারি
LEFT JOINবাম টেবিলের সব + মিলে যাওয়া
RIGHT JOINডান টেবিলের সব + মিলে যাওয়া
CROSS JOINসব সম্ভাব্য সংমিশ্রণ
🔴 উন্নত

Subquery (উপ-প্রশ্ন)

Subquery হলো একটি Query এর ভেতরে আরেকটি Query। জটিল প্রশ্নের উত্তর দিতে এটি ব্যবহার হয়।

WHERE এ Subquery

SQL
-- গড়ের উপরে নম্বর পাওয়া ছাত্র
SELECT name, marks
FROM students
WHERE marks > (
  SELECT AVG(marks) FROM students
);

-- সর্বোচ্চ নম্বর পাওয়া ছাত্র
SELECT * FROM students
WHERE marks = (
  SELECT MAX(marks) FROM students
);

FROM এ Subquery (Derived Table)

SQL
-- ক্লাসভিত্তিক গড় থেকে শীর্ষ ক্লাস
SELECT *
FROM (
  SELECT class, AVG(marks) AS গড়
  FROM students
  GROUP BY class
) AS class_avg
WHERE গড় > 80;

EXISTS Subquery

SQL
-- যাদের teacher_id আছে
SELECT name FROM students s
WHERE EXISTS (
  SELECT 1 FROM teachers t
  WHERE t.id = s.teacher_id
);
🔴 উন্নত

Index ও Performance

Index হলো বইয়ের সূচিপত্রের মতো — এটি ডেটা দ্রুত খুঁজে পেতে সাহায্য করে।

Index তৈরি করো

SQL
-- সাধারণ Index
CREATE INDEX idx_name ON students(name);

-- Unique Index
CREATE UNIQUE INDEX idx_roll ON students(roll);

-- একাধিক কলামে Index
CREATE INDEX idx_class_marks ON students(class, marks);

-- Index দেখো
SHOW INDEX FROM students;

-- Index মুছো
DROP INDEX idx_name ON students;

EXPLAIN দিয়ে Query বিশ্লেষণ করো

SQL
-- Query কীভাবে চলছে দেখো
EXPLAIN SELECT * FROM students
WHERE name = 'আরিফ হোসেন';
📌 Performance টিপস:
✅ WHERE, JOIN, ORDER BY তে ব্যবহৃত কলামে Index দাও
✅ SELECT * এর বদলে শুধু দরকারি কলাম ব্যবহার করো
✅ LIMIT ব্যবহার করো বড় ডেটাসেটে
✅ অপ্রয়োজনীয় Index এড়িয়ে চলো — এটি INSERT/UPDATE ধীর করে
🔴 উন্নত

View ও Stored Procedure

View হলো সংরক্ষিত Query। Stored Procedure হলো ডেটাবেসে সংরক্ষিত কোডের সমষ্টি।

View তৈরি করো

SQL
-- View তৈরি করো
CREATE VIEW top_students AS
SELECT name, class, marks
FROM students
WHERE marks >= 90
ORDER BY marks DESC;

-- View ব্যবহার করো (টেবিলের মতো)
SELECT * FROM top_students;

-- View মুছো
DROP VIEW top_students;

Stored Procedure তৈরি করো

SQL
DELIMITER //

CREATE PROCEDURE GetStudentsByClass(
  IN p_class VARCHAR(20)
)
BEGIN
  SELECT name, roll, marks
  FROM students
  WHERE class = p_class
  ORDER BY marks DESC;
END //

DELIMITER ;

-- Procedure চালাও
CALL GetStudentsByClass('দশম');
💡 View এর সুবিধা: জটিল Query বারবার লিখতে হয় না। নিরাপত্তার জন্য সরাসরি টেবিল না দেখিয়ে View দেওয়া যায়।
🔴 উন্নত

Transaction ও Security

Transaction নিশ্চিত করে যে একাধিক অপারেশন হয় সবগুলো সফল হবে, নয়তো কোনোটিই হবে না।

Transaction ব্যবহার

SQL
-- Transaction শুরু করো
START TRANSACTION;

-- অপারেশন করো
UPDATE accounts SET balance = balance - 1000
WHERE id = 1;

UPDATE accounts SET balance = balance + 1000
WHERE id = 2;

-- সফল হলে সংরক্ষণ করো
COMMIT;

-- সমস্যা হলে বাতিল করো
ROLLBACK;

User ও Permission

SQL
-- নতুন User তৈরি করো
CREATE USER 'rahim'@'localhost'
  IDENTIFIED BY 'শক্তিশালী_পাসওয়ার্ড';

-- Permission দাও
GRANT SELECT, INSERT, UPDATE ON
  school_db.* TO 'rahim'@'localhost';

-- সব Permission দাও (Admin)
GRANT ALL PRIVILEGES ON *.* TO 'admin'@'localhost';

-- Permission প্রয়োগ করো
FLUSH PRIVILEGES;

-- Permission সরাও
REVOKE INSERT ON school_db.* FROM 'rahim'@'localhost';

-- User মুছো
DROP USER 'rahim'@'localhost';

Isolation — Transaction পরস্পর থেকে আলাদা
Durability — Commit হলে স্থায়ী
